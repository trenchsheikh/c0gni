## SDK Overview

### Python SDK

```python
from c0gni import BaseAgent, MarketData, Decision, TradeResult

class CustomSniperAgent(BaseAgent):
    def __init__(self, config):
        super().__init__(config)
        self.min_liquidity = config.get('min_liquidity', 100000)
        self.social_weight = config.get('social_weight', 0.3)
        
    async def analyze_opportunity(self, opportunity):
        """Custom analysis logic"""
        
        # Technical analysis
        technical_score = self.calculate_technical_score(opportunity)
        
        # Social sentiment analysis
        social_score = await self.get_social_sentiment(opportunity.token)
        
        # Liquidity analysis
        liquidity_score = self.assess_liquidity(opportunity)
        
        # Combined scoring
        final_score = (
            technical_score * 0.4 +
            social_score * self.social_weight +
            liquidity_score * 0.3
        )
        
        return Decision(
            action='BUY' if final_score > 0.7 else 'PASS',
            confidence=final_score,
            reasoning=f"Technical: {technical_score:.2f}, Social: {social_score:.2f}, Liquidity: {liquidity_score:.2f}"
        )
    
    async def execute_trade(self, decision, opportunity):
        """Custom execution logic"""
        if decision.action == 'BUY':
            position_size = self.calculate_position_size(decision.confidence)
            
            return await self.submit_trade({
                'token': opportunity.token,
                'side': 'BUY',
                'amount': position_size,
                'slippage_tolerance': 0.05,
                'use_jito': True,
                'priority_fee': 'AUTO'
            })
```

### TypeScript SDK

```typescript
import { BaseAgent, MarketData, Decision, AgentConfig } from '@c0gni/sdk';

export class ArbitrageAgent extends BaseAgent {
  private exchanges = ['Jupiter', 'Raydium', 'Orca'];
  private minSpread: number;
  
  constructor(config: AgentConfig) {
    super(config);
    this.minSpread = config.minSpread || 0.005; // 0.5% minimum
  }
  
  async findArbitrageOpportunities(): Promise<ArbitrageOpportunity[]> {
    const opportunities: ArbitrageOpportunity[] = [];
    
    for (const token of this.monitoredTokens) {
      const prices = await Promise.all(
        this.exchanges.map(exchange => this.getPrice(token, exchange))
      );
      
      const maxPrice = Math.max(...prices);
      const minPrice = Math.min(...prices);
      const spread = (maxPrice - minPrice) / minPrice;
      
      if (spread > this.minSpread) {
        opportunities.push({
          token,
          buyExchange: this.exchanges[prices.indexOf(minPrice)],
          sellExchange: this.exchanges[prices.indexOf(maxPrice)],
          spread,
          profitEstimate: this.calculateProfit(spread, token)
        });
      }
    }
    
    return opportunities.sort((a, b) => b.spread - a.spread);
  }
  
  async executeArbitrage(opportunity: ArbitrageOpportunity): Promise<TradeResult> {
    // Execute simultaneous buy/sell orders
    const [buyResult, sellResult] = await Promise.all([
      this.placeBuyOrder(opportunity),
      this.placeSellOrder(opportunity)
    ]);
    
    return this.combineResults(buyResult, sellResult);
  }
}
```

    ```python
    from c0gni.simulation import BacktestEngine
    
    class BacktestRunner:
        def __init__(self, agent_class, config):
            self.engine = BacktestEngine()
            self.agent = agent_class(config)
            
        async def run_backtest(self, start_date, end_date):
            # Load historical market data
            market_data = await self.engine.load_data(start_date, end_date)
            
            results = BacktestResults()
            
            for timestamp, data in market_data:
                # Simulate market conditions at this timestamp
                self.engine.set_timestamp(timestamp)
                
                # Let agent analyze and make decisions
                opportunities = self.agent.scan_opportunities(data)
                
                for opp in opportunities:
                    decision = await self.agent.analyze_opportunity(opp)
                    
                    if decision.action == 'BUY':
                        trade = await self.engine.simulate_trade(decision, opp)
                        results.add_trade(trade)
            
            return results.generate_report()
    ```

    **Custom TA Library**
    
    ```python
    from c0gni.indicators import IndicatorBase
    
    class CustomRSI(IndicatorBase):
        def __init__(self, period=14, overbought=70, oversold=30):
            self.period = period
            self.overbought = overbought
            self.oversold = oversold
            
        def calculate(self, prices):
            """Calculate RSI with custom logic"""
            deltas = np.diff(prices)
            gains = np.where(deltas > 0, deltas, 0)
            losses = np.where(deltas < 0, -deltas, 0)
            
            avg_gain = np.mean(gains[-self.period:])
            avg_loss = np.mean(losses[-self.period:])
            
            rs = avg_gain / avg_loss if avg_loss != 0 else 100
            rsi = 100 - (100 / (1 + rs))
            
            return {
                'value': rsi,
                'signal': self.get_signal(rsi),
                'strength': abs(rsi - 50) / 50
            }
            
        def get_signal(self, rsi):
            if rsi > self.overbought:
                return 'SELL'
            elif rsi < self.oversold:
                return 'BUY'
            else:
                return 'NEUTRAL'
    ```

    ```typescript
    import { SentimentAnalyzer, SocialDataProvider } from '@c0gni/social';
    
    class SocialSentimentAgent extends BaseAgent {
      private sentiment: SentimentAnalyzer;
      private social: SocialDataProvider;
      
      constructor(config: AgentConfig) {
        super(config);
        this.sentiment = new SentimentAnalyzer(config.sentimentConfig);
        this.social = new SocialDataProvider(config.socialConfig);
      }
      
      async analyzeSocialSentiment(token: string): Promise<SentimentScore> {
        // Gather social media mentions
        const mentions = await this.social.getMentions(token, {
          sources: ['twitter', 'discord', 'telegram'],
          timeframe: '24h',
          minFollowers: 1000
        });
        
        // Analyze sentiment
        const scores = await Promise.all(
          mentions.map(mention => this.sentiment.analyze(mention.text))
        );
        
        // Weighted average by influence
        const weightedScore = this.calculateWeightedSentiment(scores, mentions);
        
        return {
          score: weightedScore,
          confidence: this.calculateConfidence(scores),
          volume: mentions.length,
          trending: this.detectTrend(mentions)
        };
      }
    }
    ```

    ```python
    from c0gni.deployment import ProductionDeployer
    
    class ProductionSetup:
        def deploy_agent(self, agent_code, config):
            deployer = ProductionDeployer()
            
            # Validate agent code
            validation = deployer.validate_agent(agent_code)
            if not validation.passed:
                raise ValueError(f"Validation failed: {validation.errors}")
            
            # Deploy to production
            deployment = deployer.deploy(
                agent_code=agent_code,
                config=config,
                environment='production',
                monitoring_enabled=True
            )
            
            return deployment
    ```
